> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Importar arquivos de mídia (menu, Ctrl+I, arrastar), listá-los no Media Pool com thumbnail e metadados, e exibir qualquer quadro do clipe selecionado no viewer de fonte com precisão de quadro e timecode.

Testes geram a própria mídia sintética com o FFmpeg linkado (encoder `mpeg4` + `pcm_s16le` em `.mov`), cujo brilho de cada quadro codifica o número do quadro.

**Tech Stack:** C++20, Qt 6.11.3 (Core, Gui, Widgets, Concurrent, Test), FFmpeg 9.0.2 (avformat, avcodec, swscale, avutil) via vcpkg.

**Spec:** [docs/superpowers/specs/2026-10-13-mcs-ai-editor-design.md](../specs/2026-10-13-mcs-ai-editor-design.md) — marco v0.2 (seção 9), pipeline de mídia (seção 6 itens 1–3), modelo (seção 7), interface (seção 8), testes (seção 10).

## Decisões deste plano (com a spec como autoridade)

- **Viewer v0.2 = `QPainter` (CPU).** A spec põe viewers em `QRhiWidget` sobre o grafo de render GPU; o grafo e o QRhi entram no v0.3 junto com o playback. O v0.2 prova import/probe/decode exato; misturar GPU aqui somaria dois riscos num marco. O `FrameView` é trocado no v0.3.
- **Decode por software no v0.2.** D3D11VA (spec seção 6.2) entra no v0.3, onde playback contínuo exige; scrub de quadro parado não exige.
- **Probe e decode síncronos; thumbnails assíncronos.** Probe lê só cabeçalhos (rápido). Thumbnail decodifica → QtConcurrent. Decode no scrub é síncrono no v0.2 (cache/prefetch são do v0.3).
- **Timecode non-drop-frame** no v0.2 (29,97 usa base nominal 30). Drop-frame fica para quando houver timeline (v0.3+).

```cpp
namespace mcs::core {

// Fração exata (ex.: fps 30000/1001). Nunca use double para posições de tempo.
struct Rational {
    std::int64_t num = 0;
    std::int64_t den = 1;

    double toDouble() const { return den == 0 ? 0.0 : static_cast<double>(num) / den; }
    friend bool operator==(const Rational&, const Rational&) = default;
};
```

```cpp
#include "core/rational.h"

#include <cmath>
#include <numeric>

namespace mcs::core {

Rational normalized(Rational r) {
    if (r.den == 0) {
        return {0, 1};
    }
    if (r.den < 0) {
        r.num = -r.num;
        r.den = -r.den;
    }
    const std::int64_t g = std::gcd(r.num, r.den);
    return g > 1 ? Rational{r.num / g, r.den / g} : r;
}

QString timecode(std::int64_t frame, Rational fps) {
    const auto base = static_cast<std::int64_t>(std::llround(fps.toDouble()));
    if (base <= 0 || frame < 0) {
        return QStringLiteral("--:--:--:--");
    }
    const std::int64_t ff = frame % base;
    const std::int64_t totalSeconds = frame / base;
    const std::int64_t ss = totalSeconds % 60;
    const std::int64_t mm = (totalSeconds / 60) % 60;
    const std::int64_t hh = totalSeconds / 3600;
    return QStringLiteral("%1:%2:%3:%4")
        .arg(hh, 2, 10, QLatin1Char('0'))
        .arg(mm, 2, 10, QLatin1Char('0'))
        .arg(ss, 2, 10, QLatin1Char('0'))
        .arg(ff, 2, 10, QLatin1Char('0'));
}

} // namespace mcs::core
```

```cpp
#include "support/synth_media.h"

#include <QScopeGuard>
#include <algorithm>
#include <cmath>
#include <cstdint>
#include <cstring>

extern "C" {
#include <libavcodec/avcodec.h>
#include <libavformat/avformat.h>
#include <libavutil/channel_layout.h>
}

namespace mcs::test {

int lumaForFrame(int k) { return 16 + (k * 7) % 200; }

int expectedGrayForFrame(int k) {
    const double v = (lumaForFrame(k) - 16) * 255.0 / 219.0;
    return static_cast<int>(std::lround(std::clamp(v, 0.0, 255.0)));
}

QString writeSynthMov(const QString& path, const SynthSpec& s) {
    const QByteArray file = path.toUtf8();
    AVFormatContext* oc = nullptr;
    if (avformat_alloc_output_context2(&oc, nullptr, "mov", file.constData()) < 0 || !oc) {
        return QStringLiteral("avformat_alloc_output_context2 falhou");
    }
    AVCodecContext* vc = nullptr;
    AVCodecContext* ac = nullptr;
    AVPacket* pkt = av_packet_alloc();
    AVFrame* f = av_frame_alloc();
    auto cleanup = qScopeGuard([&] {
        av_frame_free(&f);
        av_packet_free(&pkt);
        avcodec_free_context(&vc);
        avcodec_free_context(&ac);
        if (oc->pb) {
            avio_closep(&oc->pb);
        }
        avformat_free_context(oc);
    });
    const bool globalHeader = (oc->oformat->flags & AVFMT_GLOBALHEADER) != 0;

    AVStream* vs = nullptr;
    if (s.video) {
        const AVCodec* codec = avcodec_find_encoder(AV_CODEC_ID_MPEG4);
        if (!codec) {
            return QStringLiteral("encoder mpeg4 ausente no FFmpeg");
        }
        vc = avcodec_alloc_context3(codec);
        vc->width = s.width;
        vc->height = s.height;
        vc->pix_fmt = AV_PIX_FMT_YUV420P;
        vc->time_base = AVRational{1, s.fps};
        vc->framerate = AVRational{s.fps, 1};
        vc->gop_size = s.gop;
        vc->max_b_frames = 0;
        vc->bit_rate = 2'000'000;
        if (globalHeader) {
            vc->flags |= AV_CODEC_FLAG_GLOBAL_HEADER;
        }
        if (avcodec_open2(vc, codec, nullptr) < 0) {
            return QStringLiteral("avcodec_open2(mpeg4) falhou");
        }
        vs = avformat_new_stream(oc, nullptr);
        avcodec_parameters_from_context(vs->codecpar, vc);
        vs->time_base = vc->time_base;
    }

    AVStream* as = nullptr;
    if (s.audio) {
        const AVCodec* codec = avcodec_find_encoder(AV_CODEC_ID_PCM_S16LE);
        if (!codec) {
            return QStringLiteral("encoder pcm_s16le ausente no FFmpeg");
        }
        ac = avcodec_alloc_context3(codec);
        ac->sample_fmt = AV_SAMPLE_FMT_S16;
        ac->sample_rate = 48000;
        av_channel_layout_default(&ac->ch_layout, 2);
        ac->time_base = AVRational{1, 48000};
        if (globalHeader) {
            ac->flags |= AV_CODEC_FLAG_GLOBAL_HEADER;
        }
        if (avcodec_open2(ac, codec, nullptr) < 0) {
            return QStringLiteral("avcodec_open2(pcm_s16le) falhou");
        }
        as = avformat_new_stream(oc, nullptr);
        avcodec_parameters_from_context(as->codecpar, ac);
        as->time_base = ac->time_base;
    }

    if (!(oc->oformat->flags & AVFMT_NOFILE) &&
        avio_open(&oc->pb, file.constData(), AVIO_FLAG_WRITE) < 0) {
        return QStringLiteral("avio_open falhou: %1").arg(path);
    }
    if (avformat_write_header(oc, nullptr) < 0) {
        return QStringLiteral("avformat_write_header falhou");
    }

    auto drain = [&](AVCodecContext* c, AVStream* st) {
        for (;;) {
            const int r = avcodec_receive_packet(c, pkt);
            if (r == AVERROR(EAGAIN) || r == AVERROR_EOF) {
                return true;
            }
            if (r < 0) {
                return false;
            }
            av_packet_rescale_ts(pkt, c->time_base, st->time_base);
            pkt->stream_index = st->index;
            if (av_interleaved_write_frame(oc, pkt) < 0) {
                return false;
            }
        }
    };

    const int samplesPerFrame = 48000 / s.fps;
    for (int k = 0; k < s.frames; ++k) {
        if (vc) {
            av_frame_unref(f);
            f->format = vc->pix_fmt;
            f->width = vc->width;
            f->height = vc->height;
            if (av_frame_get_buffer(f, 0) < 0) {
                return QStringLiteral("av_frame_get_buffer(video) falhou");
            }
            const auto y = static_cast<unsigned char>(lumaForFrame(k));
            for (int row = 0; row < f->height; ++row) {
                std::memset(f->data[0] + row * f->linesize[0], y, static_cast<size_t>(f->width));
            }
            const auto chromaW = static_cast<size_t>((f->width + 1) / 2);
            for (int row = 0; row < (f->height + 1) / 2; ++row) {
                std::memset(f->data[1] + row * f->linesize[1], 128, chromaW);
                std::memset(f->data[2] + row * f->linesize[2], 128, chromaW);
            }
            f->pts = k;
            if (avcodec_send_frame(vc, f) < 0 || !drain(vc, vs)) {
                return QStringLiteral("encode de vídeo falhou no quadro %1").arg(k);
            }
        }
        if (ac) {
            av_frame_unref(f);
            f->format = ac->sample_fmt;
            f->sample_rate = ac->sample_rate;
            av_channel_layout_copy(&f->ch_layout, &ac->ch_layout);
            f->nb_samples = samplesPerFrame;
            if (av_frame_get_buffer(f, 0) < 0) {
                return QStringLiteral("av_frame_get_buffer(audio) falhou");
            }
            std::memset(f->data[0], 0, static_cast<size_t>(samplesPerFrame) * 2 * 2);
            f->pts = static_cast<std::int64_t>(k) * samplesPerFrame;
            if (avcodec_send_frame(ac, f) < 0 || !drain(ac, as)) {
                return QStringLiteral("encode de áudio falhou no quadro %1").arg(k);
            }
        }
    }
    if (vc && (avcodec_send_frame(vc, nullptr) < 0 || !drain(vc, vs))) {
        return QStringLiteral("flush de vídeo falhou");
    }
    if (ac && (avcodec_send_frame(ac, nullptr) < 0 || !drain(ac, as))) {
        return QStringLiteral("flush de áudio falhou");
    }
    if (av_write_trailer(oc) < 0) {
        return QStringLiteral("av_write_trailer falhou");
    }
    return {};
}

} // namespace mcs::test
```

```cpp
core::Rational frameRateOf(AVFormatContext* fmt, AVStream* st) {
    const AVRational r = av_guess_frame_rate(fmt, st, nullptr);
    return core::normalized({r.num, r.den});
}

std::int64_t frameCountOf(AVFormatContext* fmt, AVStream* st, core::Rational fps) {
    if (st->nb_frames > 0) {
        return st->nb_frames;
    }
    if (fps.num <= 0) {
        return 0;
    }
    const AVRational frameTb{static_cast<int>(fps.den), static_cast<int>(fps.num)};
    if (st->duration != AV_NOPTS_VALUE) {
        return av_rescale_q(st->duration, st->time_base, frameTb);
    }
    if (fmt->duration != AV_NOPTS_VALUE) {
        return av_rescale_q(fmt->duration, AVRational{1, AV_TIME_BASE}, frameTb);
    }
    return 0;
}

    return name ? QString::fromUtf8(name) : QStringLiteral("unknown");
```

```cpp
        vi.colorTransfer = detail::nameOr(av_color_transfer_name(p->color_trc));
        vi.colorMatrix = detail::nameOr(av_color_space_name(p->color_space));
        vi.colorRange = detail::nameOr(av_color_range_name(p->color_range));
        info.video = vi;
    }

    const int a = av_find_best_stream(fmt, AVMEDIA_TYPE_AUDIO, -1, -1, nullptr, 0);
    if (a >= 0) {
        const AVCodecParameters* p = fmt->streams[a]->codecpar;
        core::AudioStreamInfo ai;
        ai.sampleRate = p->sample_rate;
        ai.channels = p->ch_layout.nb_channels;
        ai.codec = detail::nameOr(avcodec_get_name(p->codec_id));
        info.audio = ai;
```

```cpp
extern "C" {
#include <libavcodec/avcodec.h>
#include <libavformat/avformat.h>
#include <libswscale/swscale.h>
}

namespace mcs::media {

struct FrameDecoder::Impl {
    AVFormatContext* fmt = nullptr;
    AVCodecContext* dec = nullptr;
    SwsContext* sws = nullptr;
    AVFrame* frame = nullptr;
    AVPacket* pkt = nullptr;
    AVStream* st = nullptr;
    int streamIndex = -1;
    core::Rational fps;
    std::int64_t count = 0;
    std::int64_t startPts = 0;

    ~Impl() {
        sws_freeContext(sws);
        av_frame_free(&frame);
        av_packet_free(&pkt);
        avcodec_free_context(&dec);
        avformat_close_input(&fmt);
    }

    AVRational frameTb() const {
        return AVRational{static_cast<int>(fps.den), static_cast<int>(fps.num)};
    }

    QImage toImage(const AVFrame* f) {
        sws = sws_getCachedContext(sws, f->width, f->height, static_cast<AVPixelFormat>(f->format),
                                   f->width, f->height, AV_PIX_FMT_RGBA, SWS_BILINEAR, nullptr,
                                   nullptr, nullptr);
        if (!sws) {
            return {};
        }
        QImage img(f->width, f->height, QImage::Format_RGBA8888);
        uint8_t* dst[4] = {img.bits(), nullptr, nullptr, nullptr};
        const int dstStride[4] = {static_cast<int>(img.bytesPerLine()), 0, 0, 0};
        sws_scale(sws, f->data, f->linesize, 0, f->height, dst, dstStride);
        return img;
    }
};

FrameDecoder::FrameDecoder() = default;
FrameDecoder::~FrameDecoder() = default;

bool FrameDecoder::open(const QString& path, QString* error) {
    d_ = std::make_unique<Impl>();
    auto fail = [&](const char* text) {
        if (error) {
            *error = QCoreApplication::translate("mcs::media", text).arg(QFileInfo(path).fileName());
        }
        d_.reset();
        return false;
    };
    const QByteArray utf8 = path.toUtf8();
    if (avformat_open_input(&d_->fmt, utf8.constData(), nullptr, nullptr) < 0) {
        return fail("Cannot open \"%1\" as media.");
    }
    if (avformat_find_stream_info(d_->fmt, nullptr) < 0) {
        return fail("\"%1\" has no readable streams.");
    }
    const AVCodec* codec = nullptr;
    d_->streamIndex = av_find_best_stream(d_->fmt, AVMEDIA_TYPE_VIDEO, -1, -1, &codec, 0);
    if (d_->streamIndex < 0 || !codec) {
        return fail("\"%1\" has no video.");
    }
    d_->st = d_->fmt->streams[d_->streamIndex];
    d_->dec = avcodec_alloc_context3(codec);
    if (!d_->dec || avcodec_parameters_to_context(d_->dec, d_->st->codecpar) < 0) {
        return fail("Cannot prepare the decoder for \"%1\".");
    }
    d_->dec->thread_count = 0; // automático
    if (avcodec_open2(d_->dec, codec, nullptr) < 0) {
        return fail("Cannot open the decoder for \"%1\".");
    }
    d_->fps = detail::frameRateOf(d_->fmt, d_->st);
    if (d_->fps.num <= 0) {
        return fail("\"%1\" has no valid frame rate.");
    }
    d_->count = detail::frameCountOf(d_->fmt, d_->st, d_->fps);
    d_->startPts = d_->st->start_time != AV_NOPTS_VALUE ? d_->st->start_time : 0;
    d_->frame = av_frame_alloc();
    d_->pkt = av_packet_alloc();
    return true;
}

bool FrameDecoder::isOpen() const { return d_ != nullptr; }

std::int64_t FrameDecoder::frameCount() const { return d_ ? d_->count : 0; }

core::Rational FrameDecoder::fps() const { return d_ ? d_->fps : core::Rational{}; }

QSize FrameDecoder::size() const {
    return d_ ? QSize(d_->st->codecpar->width, d_->st->codecpar->height) : QSize();
}

std::optional<DecodedFrame> FrameDecoder::frameAt(std::int64_t index) {
    if (!d_ || index < 0 || index >= d_->count) {
        return std::nullopt;
    }
    Impl& s = *d_;
    const std::int64_t target = s.startPts + av_rescale_q(index, s.frameTb(), s.st->time_base);
    if (av_seek_frame(s.fmt, s.streamIndex, target, AVSEEK_FLAG_BACKWARD) < 0) {
        return std::nullopt;
    }
    avcodec_flush_buffers(s.dec);

    bool draining = false;
    for (;;) {
        const int r = avcodec_receive_frame(s.dec, s.frame);
        if (r == 0) {
            std::int64_t pts = s.frame->best_effort_timestamp;
            if (pts == AV_NOPTS_VALUE) {
                pts = s.frame->pts;
            }
            const std::int64_t got =
                av_rescale_q_rnd(pts - s.startPts, s.st->time_base, s.frameTb(),
                                 static_cast<AVRounding>(AV_ROUND_NEAR_INF));
            if (got >= index) {
                DecodedFrame out{got, s.toImage(s.frame)};
                av_frame_unref(s.frame);
                if (out.image.isNull()) {
                    return std::nullopt;
                }
                return out;
            }
            av_frame_unref(s.frame);
            continue;
        }
        // EOF, erro, ou decoder já esvaziado sem chegar ao quadro: não há quadro.
        if (r != AVERROR(EAGAIN) || draining) {
            return std::nullopt;
        }
        if (av_read_frame(s.fmt, s.pkt) < 0) {
            avcodec_send_packet(s.dec, nullptr); // esvazia o decoder
            draining = true;
            continue;
        }
        if (s.pkt->stream_index == s.streamIndex) {
            avcodec_send_packet(s.dec, s.pkt);
        }
        av_packet_unref(s.pkt);
    }
}

} // namespace mcs::media
```

```cpp
namespace mcs::ui {

MediaPoolModel::MediaPoolModel(QObject* parent) : QAbstractListModel(parent) {}

int MediaPoolModel::rowCount(const QModelIndex& parent) const {
    return parent.isValid() ? 0 : static_cast<int>(items_.size());
}

QVariant MediaPoolModel::data(const QModelIndex& index, int role) const {
    if (!index.isValid() || index.row() < 0 || index.row() >= items_.size()) {
        return {};
    }
    const Item& it = items_.at(index.row());
    switch (role) {
    case Qt::DisplayRole:
        return QFileInfo(it.info.path).fileName();
    case Qt::DecorationRole:
        return it.thumbnail;
    case PathRole:
        return it.info.path;
    case Qt::ToolTipRole: {
        QStringList parts;
        if (it.info.video) {
            const auto& v = *it.info.video;
            parts << QStringLiteral("%1x%2").arg(v.width).arg(v.height)
                  << tr("%1 fps").arg(v.fps.toDouble(), 0, 'f', 3)
                  << core::timecode(v.frameCount, v.fps) << v.codec;
        }
        if (it.info.audio) {
            const auto& a = *it.info.audio;
            parts << tr("%1 Hz, %2 ch").arg(a.sampleRate).arg(a.channels) << a.codec;
        }
        return parts.join(QStringLiteral(" · "));
    }
    default:
        return {};
    }
}

bool MediaPoolModel::contains(const QString& path) const {
    const QString key = QFileInfo(path).absoluteFilePath();
    for (const Item& it : items_) {
        if (QFileInfo(it.info.path).absoluteFilePath().compare(key, Qt::CaseInsensitive) == 0) {
            return true;
        }
    }
    return false;
}

QStringList MediaPoolModel::importFiles(const QStringList& paths) {
    QStringList errors;
    for (const QString& path : paths) {
        if (contains(path)) {
            continue;
        }
        const media::ProbeResult r = media::probe(path);
        if (!r.info) {
            errors << r.error;
            continue;
        }
        const int row = static_cast<int>(items_.size());
        beginInsertRows({}, row, row);
        items_.append(Item{QUuid::createUuid(), *r.info, {}});
        endInsertRows();
        requestThumbnail(items_.last().id, path);
    }
    return errors;
}

const MediaPoolModel::Item& MediaPoolModel::item(int row) const { return items_.at(row); }

void MediaPoolModel::requestThumbnail(const QUuid& id, const QString& path) {
    auto* watcher = new QFutureWatcher<QImage>(this);
    connect(watcher, &QFutureWatcher<QImage>::finished, this, [this, watcher, id] {
        const QImage thumb = watcher->result();
        watcher->deleteLater();
        for (int row = 0; row < items_.size(); ++row) {
            if (items_[row].id == id) {
                items_[row].thumbnail = thumb;
                const QModelIndex idx = index(row);
                emit dataChanged(idx, idx, {Qt::DecorationRole});
                emit thumbnailReady(row);
                return;
            }
        }
    });
    watcher->setFuture(QtConcurrent::run([path]() -> QImage {
        media::FrameDecoder decoder;
        if (!decoder.open(path)) {
            return {};
        }
        const auto frame = decoder.frameAt(0);
        return frame ? frame->image.scaled(kThumbnailSize, Qt::KeepAspectRatio,
                                           Qt::SmoothTransformation)
                     : QImage{};
    }));
}
```

```cpp
bool EditPage::eventFilter(QObject* watched, QEvent* event) {
    if (watched == list_->viewport()) {
        if (event->type() == QEvent::DragEnter) {
            auto* e = static_cast<QDragEnterEvent*>(event);
            if (e->mimeData()->hasUrls()) {

                    paths << url.toLocalFile();
                }
            }
            const QStringList errors = pool_->importFiles(paths);
            if (!errors.isEmpty()) {
                emit importFailed(errors);
            }
            e->acceptProposedAction();
            return true;
        }
    }
    return QWidget::eventFilter(watched, event);
}

} // namespace mcs::ui
```
